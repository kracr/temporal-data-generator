PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX ace: <https://anonymous.com/AcademicConferenceEvent#>
PREFIX tweet: <https://anonymous.com/Tweet#>

SELECT ?name WHERE { 
?tweet tweet:isAbout ?phase.
  BIND(
    IF(CONTAINS(STR(?phase), "#"), 
       STRAFTER(STR(?phase), "#"), 
       STRAFTER(STR(?phase), "/")
    ) AS ?name
  )
}
